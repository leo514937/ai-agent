```python
# 改造后结构
def _h_target_resolve(state):
    """薄调度层，按 candidate_source 分流"""
    task_type = state.get("task_type")
    candidate_source = state.get("candidate_source", "")
    
    if task_type in (TaskType.recommendation,):
        return _resolve_for_recommendation(state)
    
    if task_type == TaskType.comparison:
        if candidate_source == "explicit":
            return _resolve_explicit_targets(state)
        elif candidate_source == "context":
            return _resolve_context_targets(state)
        elif candidate_source == "discovery":
            return _resolve_discovery_targets(state)
        elif candidate_source == "mixed":
            return _resolve_mixed_targets(state)
        else:
            return _resolve_unclear_comparison(state)
    
    if task_type in (TaskType.single_shop_query, TaskType.coupon_query):
        return _resolve_single_shop(state)
    
    return {"resolve_shop_result": ResolveShopResult(status="NOT_FOUND")}
```
